code:

    printf("Original array: ");
    for(int i = 0; i < 5; i++)
    {
        printf("%d ", *(ptr + i));
    }

2. Calculate string length using pointer

code:

#include <stdio.h>
int main()
{
    char str[100];
    char *p;
    int length = 0;
    printf("Enter a string: ");
    scanf("%s", str);
    p = str;
    while(*p != '\0')
    {
        length++;
        p++;
    }
    printf("Length of string = %d", length);
    return 0;
}

#include <stdlib.h>

{

    float mark;
};

{
    struct Student *p;

7.Sort an array of floating point numbers using Selection Sort.

code:

#include <stdio.h>
int main()
{
    float a[5];
    int i, j, min;
    float temp;
    printf("Enter 5 floating point numbers:\n");
    for(i = 0; i < 5; i++)
        scanf("%f", &a[i]);
    for(i = 0; i < 4; i++)
    {
        min = i;
        for(j = i + 1; j < 5; j++)
        {
            if(a[j] < a[min])
                min = j;
        }
        temp = a[i];
        a[i] = a[min];
        a[min] = temp;
    }
    printf("Sorted array:\n");

    for(i = 0; i < 5; i++)
        printf("%.2f ", a[i]);
    return 0;
}

output:

Enter 5 floating point numbers:
1.6 2.4 3.6 4.2 5.9
Sorted array:
1.60 2.40 3.60 4.20 5.90

9.Sort a set of negative numbers using radix sort.

code:

#include <stdio.h>
#include <stdlib.h>
void radixSort(int a[], int n)
{
    int i, j, max, exp = 1;
    int *b = (int *)malloc(n * sizeof(int));
    max = abs(a[0]);
    for (i = 1; i < n; i++)
    {
        if (abs(a[i]) > max)
            max = abs(a[i]);
    }
    while (max / exp > 0)
    {
        int count[10] = {0};
        for (i = 0; i < n; i++)
            count[(abs(a[i]) / exp) % 10]++;
        for (i = 1; i < 10; i++)
            count[i] += count[i - 1];
        for (i = n - 1; i >= 0; i--)
            b[--count[(abs(a[i]) / exp) % 10]] = a[i];
        for (i = 0; i < n; i++)
            a[i] = b[i];
        exp *= 10;
}
    for (i = 0; i < n; i++)
        a[i] = -abs(a[i]);
    free(b);
}
int main()
{
    int a[] = {-45, -12, -89, -34, -7};
    int n = 5, i;
    radixSort(a, n);
    printf("Sorted array:\n");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);
    return 0;
}

output:

Sorted array:
-7 -12 -34 -45 -89 

10.Write a C program to copy one array to another using pointers

code:

#include <stdio.h>
int main()
{
    int a[5], b[5];
    int i;
    int *p, *q;
    printf("Enter 5 elements:\n");
    for (i = 0; i < 5; i++)
        scanf("%d", &a[i]);
    p = a;
    q = b;
    for (i = 0; i < 5; i++)
    {
        *q = *p;
        p++;
        q++;
    }
    printf("First array:\n");
    for (i = 0; i < 5; i++)
        printf("%d ", a[i]);
    printf("\nSecond array:\n");
    for (i = 0; i < 5; i++)
        printf("%d ", b[i]);
    return 0;
}
